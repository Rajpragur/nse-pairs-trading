cmake_minimum_required(VERSION 3.20)
project(nse_pairs_trading LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(pairs_benchmark benchmark.cpp)
add_executable(pairs_test test_pairs.cpp)
target_compile_options(pairs_benchmark PRIVATE -O3 -Wall -Wextra -Wpedantic)
target_compile_options(pairs_test PRIVATE -O2 -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME pairs_regression COMMAND pairs_test)
